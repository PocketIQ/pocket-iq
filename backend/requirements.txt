# CueIQ backend — everything Person A needs for the whole project.
# Install:  pip install -r requirements.txt   (inside your venv)

# --- Web server ---
fastapi                 # the API framework
uvicorn[standard]       # runs the FastAPI app
python-multipart        # lets FastAPI receive uploaded photos

# --- Computer vision ---
opencv-contrib-python-headless   # OpenCV + the ArUco marker module ("headless" = no GUI, works on servers)
numpy                            # array math OpenCV is built on

# --- AI ---
google-genai            # official Gemini SDK (shot coaching + identifying balls in photos)
httpx                   # HTTP client for the ElevenLabs text-to-speech API

# --- Database ---
psycopg[binary]         # Postgres driver for Tiger Data (games + shot history)

# --- Auth ---
pyjwt[crypto]           # verifies Auth0 login tokens sent from the frontend

# --- Config ---
python-dotenv           # loads API keys from a .env file so they never get committed

# --- Tools ---
pillow                  # only for tools/make_markers.py (printable PDF)
